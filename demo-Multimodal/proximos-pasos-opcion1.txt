import matplotlib.pyplot as plt
import re

def plot_training_loss(log_file='output.log'):
    losses = []
    epochs = []
    
    try:
        with open(log_file, 'r') as f:
            for line in f:
                # Buscar patrones tipo: "Epoch 5 - Avg Loss: 1.2345"
                if "Avg Loss:" in line:
                    parts = line.split()
                    loss = float(parts[-1])
                    # Asumiendo formato "Epoch X - ..."
                    epoch_part = line.split('-')[0].strip() # "Epoch 5"
                    epoch = int(epoch_part.split()[-1])
                    losses.append(loss)
                    epochs.append(epoch)
    except FileNotFoundError:
        print(f"❌ No se encontró el archivo {log_file}")
        return

    if not losses:
        print("⚠️ No se encontraron datos de loss en el log.")
        return

    plt.figure(figsize=(10, 5))
    plt.plot(epochs, losses, label='Training Loss (Multimodal)', color='blue', marker='o')
    plt.title('Evolución del Entrenamiento FASE 2')
    plt.xlabel('Epochs')
    plt.ylabel('Loss Total (Clasificación)')
    plt.grid(True, alpha=0.3)
    plt.legend()
    plt.savefig('training_metrics_fase2.png')
    print("✅ Gráfica guardada como training_metrics_fase2.png")

if __name__ == "__main__":

2. Evaluación del Modelo (Test Set)
¿Puedo usar evaluate_deimv2.py antiguo?
NO directamente. * Razón: El script antiguo carga un modelo DEIM puro. Tu nuevo checkpoint (checkpoint_epochX.pth o final.pth) contiene un modelo DEIMv2WithFusion (el wrapper). Si intentas cargarlo con el script antiguo, fallará por incompatibilidad de claves (state_dict keys mismatch).

Solución: Nuevo Script de Evaluación Adaptado
Necesitas un script que instancie el Wrapper antes de cargar los pesos.

Crear evaluate.py en demo-Multimodal/finetuning-simple-incremental/:

Python

import torch
from torch.utils.data import DataLoader
import sys
from pathlib import Path

# Imports necesarios (ajusta las rutas según tu estructura)
PROJECT_ROOT = Path(__file__).parent.parent.parent
DEIMV2_PATH = PROJECT_ROOT / "DEIMv2"
sys.path.insert(0, str(DEIMV2_PATH))
sys.path.insert(0, str(Path(__file__).parent))

from engine.core import YAMLConfig
from engine.data.dataset.coco_eval import CocoEvaluator
from models_utils import TextEncoder
from data import get_text_prompts
from deimv2_fusion_wrapper import build_deimv2_with_fusion

def evaluate_multimodal(checkpoint_path):
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    
    # 1. Configuración (misma que train)
    config_path = "scripts/deimv2_multimodal/configs/deimv2_industrial_defects.yml"
    
    # 2. Preparar Embeddings
    text_encoder = TextEncoder(freeze=True).to(device)
    text_embeddings = text_encoder.encode_texts(get_text_prompts(), device)
    
    # 3. Construir Modelo Wrapper
    # Nota: Pasamos el checkpoint de Fase 1 para inicializar la estructura base, 
    # luego sobreescribiremos con el checkpoint entrenado de Fase 2
    base_ckpt = "scripts/deimv2_multimodal/outputs/deimv2_1024_300epochs/best_stg1.pth"
    
    print("🏗️ Construyendo arquitectura multimodal...")
    model = build_deimv2_with_fusion(base_ckpt, config_path, text_embeddings, device)
    
    # 4. Cargar pesos entrenados en Fase 2 (Fusión + Detector)
    print(f"📂 Cargando checkpoint Fase 2: {checkpoint_path}")
    ckpt = torch.load(checkpoint_path, map_location=device)
    
    # Manejo robusto de state_dict
    state_dict = ckpt['model'] if 'model' in ckpt else ckpt
    model.load_state_dict(state_dict)
    
    model.to(device)
    model.eval()
    
    # 5. Dataloader Test/Val
    cfg = YAMLConfig(config_path)
    # Forzar batch size 1 para evaluación precisa
    cfg.yaml_cfg['val_dataloader']['total_batch_size'] = 1
    val_loader = cfg.val_dataloader 
    
    # Obtener postprocessor del config
    postprocessor = cfg.postprocessor
    
    # 6. Evaluador COCO
    # Asegúrate de que dataset.coco existe (depende de tu implementación de dataset)
    coco_evaluator = CocoEvaluator(val_loader.dataset.coco, ['bbox'])
    
    print("🚀 Iniciando evaluación...")
    for i, (images, targets) in enumerate(val_loader):
        images = images.to(device)
        targets = [{k: v.to(device) for k, v in t.items()} for t in targets]
        
        with torch.no_grad():
            # Forward
            outputs = model(images)
            
        # Formatear outputs para COCO Eval
        orig_target_sizes = torch.stack([t["orig_size"] for t in targets], dim=0)
        results = postprocessor(outputs, orig_target_sizes)
        
        res = {target['image_id'].item(): output for target, output in zip(targets, results)}
        coco_evaluator.update(res)
        
        if (i+1) % 50 == 0:
            print(f"   Procesadas {i+1} imágenes...")
        
    coco_evaluator.synchronize_between_processes()
    coco_evaluator.accumulate()
    coco_evaluator.summarize()

if __name__ == "__main__":
    # Ajusta al nombre de tu mejor checkpoint
    ckpt_file = "demo-Multimodal/finetuning-simple-incremental/outputs/fase2/final.pth"
    evaluate_multimodal(ckpt_file)

Epochs: 40 puede ser poco si la convergencia es lenta. Prueba 60.
